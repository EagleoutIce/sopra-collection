    \section{The 'sopra-seraphim' class}
    \subsection{Why, what for, and wherefore?}
    \begin{frame}{Why 'sopra-seraphim'?}
        This \LaTeXe\ document class was written as part of the Sopra in the
        winter semester 2019 and summer semester 2020 and serves as
        the basis for the look of all documents of \notetext{Team 20}.
        The layout is based on \T{beamer}. For structural reasons, this documentation was prepared in the latter, so no titles are placed; the header adapts accordingly to this circumstance!\par
        To visualize the individual code snippets, the
        \T{sopra-listings} package is used. It is not relevant
        for the use of the document class.\par
        The corresponding class should also be embedded in this document: \textattachfile[subject={sopra-seraphim.cls}]{sopra-seraphim.cls}{sopra-seraphim.cls}.
    \end{frame}

    \subsection{Dependencies}
    \begin{frame}[fragile]{Dependencies}
    This package is based on \T{beamer}, as well as the following packages (all passed arguments are also
    listed, whereby an asterisk is set in the case of a value being passed. An exclamation mark before an \say{argument} declares that this is an option that controls the loading of the package):
    \begingroup\footnotesize
    \begin{multicols}{3}
        \begin{itemize}
            \def\pkgparse#1:#2\@nil{%
                \T{#1}\ifx!#2!\else\textsuperscript{(#2)}\fi%
            }
            \foreach \pkg in {tikz:,fontenc:T1, inputenc:utf8,
                babel:{english,main=ngerman},montserrat:{defaultfam,*},microtype:,nowidow:all,hyperref:hidelinks}{
                \item \expandafter\pkgparse\pkg\@nil
            }
        \end{itemize}
    \end{multicols}
    \endgroup
    All of these packages should be part of a common \LaTeX\ distribution (furthermore
    the Ti\textit{k}Z library \T{calc} is also used.
    \end{frame}


    \subsection{Installation}
    \begin{frame}[fragile]{Installation: Path Specification}
        The class is not distributed as a \T{.dtx} file, which is why the
    following options are available:
\begin{itemize}
    \item The document class can be placed in the same directory as the document.
    In this case, the inclusion instruction reads:
\begin{plainlatex}
\documentclass{sopra-seraphim}
\end{plainlatex}
    \item The document class can be placed in a subdirectory / in a
                directory shipped with the document. In
                this case, it is specified via the (relative) path:
\begin{plainlatex}
\documentclass{./Mein/Pfad/zu/sopra-seraphim}
\end{plainlatex}
\end{itemize}

    \end{frame}

    \begin{frame}[fragile]{Installation: In the texmf Tree}
        The class can be placed (using a symlink or something similar)
        into your own \emph{texmf} tree.
        For example, on Linux using texlive
        the class can be placed here: \bvoid{\~/texmf/tex/latex/}.
        The directory can be created and then updated using
        \bbash{texhash \~/texmf}. Now
        the class can be used like any other installed class:
\begin{plainlatex}
\documentclass{sopra-seraphim}
\end{plainlatex}
        \notetext{Note: for this purpose there is a Python3.5 script called \textattachfile{installer.py}{installer.py} which, when placed in the folder above, can take care of the installation/update.}
    \end{frame}

    \section{Class Configuration}
    \subsection{Accepted Parameters}
    \begin{frame}[fragile]{General Information on Parameters}
    Like most, the document class accepts arguments. In addition to
    the arguments accepted for \T{article}, the ones listed below
    can be used, whereby for arguments with a \say{counter} option
    the one active by default is written first and the other in parentheses.
    Implicitly, this means:
\begin{plainlatex}
\documentclass[setstyle]{sopra-seraphim}
\end{plainlatex}
    is called. While this document is invoked with:
\begin{plainlatex}
\documentclass[nosetstyle]{sopra-seraphim}
\end{plainlatex}
    the style of other \T{sopra-*} packages is not modified.
    \end{frame}

    \begin{frame}[fragile]{The Parameters}
        \begin{argument}{setstyle}{nosetstyle}
            With \T{setstyle}, it is automatically checked which packages of the \T{sopra-collection} (\url{https://github.com/EagleoutIce/sopra-collection}) are loaded, and the corresponding styles are set to produce a consistent layout.
        \end{argument}
    \end{frame}


    \section{Command Overview}
    \begin{frame}{Command Overview}
        As a base class, the class does not add a large set of commands for
        the user. This task is left to the extending packages.
    \end{frame}


    \subsection{Setting Data}
    \label{sec:DatenSetzen}
    \begin{frame}[fragile,allowframebreaks]{Setting Data}
        The following commands should be set in the preamble and also configure
        the metadata of the respective document (i.e. the document properties).
        They can be overwritten as often as desired (up to the point at which
        the document begins). \notetext{Developer note: all these fields are available via
        \T{\textbackslash sos@register@$\langle$Name$\rangle$}, or their 'short' variants
        via \T{\textbackslash sos@register@short@$\langle$Name$\rangle$}.}

        \begin{command}{brief}{\optArg{short}\manArg{ShortDescription}}
            Sets the short description of the document to \T{ShortDescription}. It is shown at the bottom left of the title page. \imptext{Currently not used.}
        \end{command}

        \begin{command}{authors}{\optArg{short}\manArg{AuthorList}}
            This can be used to set the list of authors. It is recommended
            to use \cmdref{addAuthor} for this.
        \end{command}

        \begin{command}{duedate}{\optArg{short}\manArg{YYYY-MM-DD}}
           Sets the date of the document. \cmdref{DateConvert} is used
           for the conversion.
        \end{command}

        \begin{command}{supervisor}{\optArg{short}\manArg{Name}}
            Sets the supervisor to \T{Name}. This data field is not used in the base class. \imptext{Currently not used.}
        \end{command}

        \begin{command}{keywords}{\optArg{short}\manArg{Keywords}}
            Sets the keywords for the metadata to \T{Keywords}.
        \end{command}

        \begin{command}{group}{\optArg{short}\manArg{Name}}
            Sets the group of the respective author to \T{Name}.% Maybe show?
        \end{command}

        \begin{command}{groupnum}{\optArg{short}\manArg{Number}}
            In case of a numbering (or symbol, etc.) of the group, sets this
            to \T{Number}.
        \end{command}
    \end{frame}


    \subsection{Helpful When Setting Data}
    \begin{frame}[fragile]{Helpful When Setting Data}
    \begin{command}{addAuthor}{\manArg{Name (Email)}}
        Adds an author to the list of authors (\cmdref{authors}). For
        example: \blatex[morekeywords={[5]{\\addAuthor}}]{\\addAuthor\{Florian Sihler (florian.sihler@uni-ulm.de)\}}.
    \end{command}

    \begin{command}{setTeam}{\manArg{Name Number}}
        Allows \cmdref{group} and \cmdref{groupnum} to be set in a single call.
        For example: \T{\cmd{setTeam}\{Affenbande 42\}} or
        \T{\cmd{setTeam}\{\{Mega Team\} A\}}.
    \end{command}
    \end{frame}


\subsection{Displaying the Data}

\begin{frame}[fragile,allowframebreaks]{Displaying the Data}
    \begin{command}{typesetAuthors}{}
        Typesets the authors according to \jmark[\T{showmail}]{mrk:showmail} with or
        without email address, so \cmd{typesetAuthors} yields: \typesetAuthors.
        With multiple authors, commas and the \say{and} are set correctly.
    \end{command}

    \begin{command}{thesosversion}{}
        Outputs the current version of the \T{sopra-seraphim} document class. \notetext{Note: via \blatex{\\value\{sosversion\}} the
        version can be obtained as a $4$-digit number: \arabic{sosversion}.}
    \end{command}

    \begin{command}{DateConvert}{\manArg{YYYY-MM-DD}}
        Converts a date into a nicer form, thus:
    \begin{plainlatex}[morekeywords={[5]{\\DateConvert}}]
\DateConvert{2019-12-02} % :yields: !*\solGet{comments}{\DateConvert{2019-12-02}}*!
    \end{plainlatex}
    \end{command}
\end{frame}

\subsection{General Formatting Commands}
\begin{frame}[fragile]{General Formatting Commands}
\begin{command}{imptext}{\manArg{Text}}
    Highlights \T{Text} as \imptext{very important}.
\end{command}

\begin{command}{notetext}{\manArg{Text}}
    Sets \T{Text} as an \notetext{incidental note}.
\end{command}
\end{frame}
